class CatalogCoverageBudgetTest(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        cls.catalog = build_catalog(ROOT)
        cls.registry = load_registry(ROOT, cls.catalog)
        cls.collection = json.loads((DATA / 'collection.json').read_bytes())
        cls.metadata, cls.metadata_files = metadata.captured()

    def event(self, identity):
        for receipt in self.collection['pages']:
            if not any(row['id'] == identity for row in receipt['records']):
                continue
            raw = gzip.decompress((DATA / receipt['raw']).read_bytes())
            self.assertEqual(hashlib.sha256(raw).hexdigest(), receipt['responseSha256'])
            return next(row for row in json.loads(raw)['results'] if row['id'] == identity)
        self.fail('actual captured launch missing: ' + identity)

    def artifact(self, row, *, catalog=None, registry=None, root=ROOT):
        # Only technical event status/time changes make historical real records
        # selectable. Display strings and typed source identities stay verbatim.
        now = NOW
        technical = deepcopy(row)
        if row['status']['id'] in (3, 4, 7):
            now = int(datetime.fromisoformat(row['net'].replace('Z', '+00:00')).timestamp()) - 3600
            technical['last_updated'] = datetime.fromtimestamp(now, timezone.utc).isoformat()
        record = replace(normalize_launch(technical, now), status=LaunchStatus.PENDING)
        state = merge_schedule(ScheduleState(), (record,), now)
        files, _ = build_artifact(state, catalog or self.catalog, now, root=root,
            eligibility=registry or self.registry,
            publisher=dict(instance='a' * 64, generation=1, selectionValidatedUtc=now))
        feed = validate_artifact(files)
        scene = loads_json(files[feed['visualSelection']['path']])
        return state, files, feed, scene

    def test_vc01_s01_positive(self):
        row = self.event('e652a538-6d40-4b55-97a6-7c757ec4e1e9')
        self.assertEqual(row['pad']['country']['alpha_2_code'], 'NO')
        self.assertEqual(row['rocket']['configuration']['full_name'], 'Spectrum')
        state, _, feed, scene = self.artifact(row)
        self.assertEqual(feed['launch']['launch_id'], 'll2:launch:e652a538-6d40-4b55-97a6-7c757ec4e1e9')
        self.assertEqual(feed['launch']['configuration_name'], 'Spectrum')
        self.assertEqual(feed['launch']['site_name'], 'Orbital Launch Pad')
        self.assertEqual(scene['modelId'], 'll2:configuration:' + str(row['rocket']['configuration']['id']))
        self.assertEqual(scene['siteId'], 'll2:site:' + str(row['pad']['id']))
        self.assertEqual(feed['launch']['source_validated_utc'], state.evaluated_at_utc)

    def test_vc04_s01_negative(self):
        # Production corpus builder refuses page omissions, duplicate IDs and
        # count drift while preserving previously accepted output bytes.
        for defect in ('omitted', 'duplicate', 'count'):
            with self.subTest(defect=defect), tempfile.TemporaryDirectory() as folder:
                path = Path(folder)
                state = deepcopy(self.collection)
                if defect == 'omitted':
                    state['pages'].pop(1)
                else:
                    page = state['pages'][1]
                    doc = json.loads(gzip.decompress((DATA / page['raw']).read_bytes()))
                    if defect == 'count':
                        doc['count'] += 1
                    else:
                        first = json.loads(gzip.decompress((DATA / state['pages'][0]['raw']).read_bytes()))
                        doc['results'][0] = first['results'][0]
                    raw = json.dumps(doc, ensure_ascii=False).encode()
                    page['responseSha256'] = hashlib.sha256(raw).hexdigest()
                    page['raw'] = 'mutated.json.gz'
                    (path / page['raw']).write_bytes(gzip.compress(raw))
                (path / 'collection.json').write_bytes(display_corpus.encoded(state))
                (path / 'raw').symlink_to(DATA / 'raw', target_is_directory=True)
                accepted = (DATA / 'display-corpus.json').read_bytes()
                (path / 'display-corpus.json').write_bytes(accepted)
                with self.assertRaises(ValueError):
                    display_corpus.build(path, root=ROOT)
                self.assertEqual((path / 'display-corpus.json').read_bytes(), accepted)

    def test_vc04_s02_negative(self):
        # A technically valid generic downgrade and missing point must appear as
        # gaps in production diagnostics and keep source strings intact.
        temporary = tempfile.TemporaryDirectory()
        self.addCleanup(temporary.cleanup)
        root = temporary_catalog(temporary.name)
        value = supplement('current-real-starship-generic-gap')
        value['updates'] = [dict(identity='ll2:configuration:522',
            changes=dict(currentPackAssetId='generic', currentPackSpecific=False),
            sources=['https://www.spacex.com/vehicles/starship/'])]
        path = root / 'catalog/real-generic-gap.json'
        path.write_bytes(canonical_bytes(value))
        catalog = amend_recipe(root, root / 'catalog/source.json', path, 'current.real.generic.gap')
        registry = load_registry(root, catalog)
        model = catalog['dictionaries']['configurations']['ll2:configuration:522']
        report = coverage_report(catalog)
        self.assertIn('ll2:configuration:522', report['configurations']['missingArtIds'])
        self.assertEqual(report['configurations']['mappedArt'], 202)
        self.assertNotEqual(report['sites']['sourceNumericPairs'], report['counts']['sites'])
        row = captured_starship()
        row['pad']['latitude'] = row['pad']['longitude'] = None
        row['pad']['id'] = 999996
        state, _, feed, scene = self.artifact(row, catalog=catalog, registry=registry, root=root)
        self.assertEqual(feed['launch']['configuration_name'], row['rocket']['configuration']['full_name'])
        self.assertIsNone(scene['position'])
        from tools.spacefox.schedule import DEFAULT_POLICY
        review = schedule_catalog_review(catalog, registry, state, NOW, DEFAULT_POLICY)
        self.assertIn('specific-visual-not-qualified', {x['reason'] for x in review['gaps']})
        # A sourced, well-formed amendment may not label generic art specific.
        false_claim = supplement('current-real-false-specific')
        false_claim['updates'] = [dict(identity='ll2:configuration:522',
            changes=dict(currentPackAssetId='generic', currentPackSpecific=True),
            sources=['https://www.spacex.com/vehicles/starship/'])]
        refused = root / 'catalog/false-specific.json'
        refused.write_bytes(canonical_bytes(false_claim))
        accepted_recipe = (root / 'catalog/source.json').read_bytes()
        with self.assertRaisesRegex(CatalogError, 'specific-art'):
            amend_recipe(root, root / 'catalog/source.json', refused, 'refused.false.specific')
        self.assertEqual((root / 'catalog/source.json').read_bytes(), accepted_recipe)

    def test_vc07_s02_positive(self):
        helper = metadata.CurrentMetadataPublisherTest()
        with helper.authority() as (pub, store, code):
            helper.primary(pub, metadata.NOW, accept=True)
            preserved = deepcopy(helper.preserved(pub))
            transport = metadata.CapturedTransport(pub, self.metadata, self.metadata_files)
            collector = CatalogRefresh(pub, transport=transport)
            self.assertEqual(collector.run(metadata.NOW, max_requests=1).requests, 1)
            prefix = deepcopy(collector.journal['staging']['pages'])
            # Real Git reopen preserves cursor and consumes just the next page.
            pub = Publisher(store, code)
            pub.reconcile(None)
            transport.pub = pub
            collector = CatalogRefresh(pub, transport=transport)
            self.assertEqual(collector.run(metadata.NOW, max_requests=4).requests, 4)
            self.assertEqual(collector.journal['staging']['pages'][:1], prefix)
            self.assertEqual(collector.run(metadata.NOW).requests, 1)
            self.assertEqual(collector.run(metadata.NOW).status, 'yielded')
            self.assertEqual([call['url'] for call in transport.calls],
                             [page['url'] for page in self.metadata['coverage']['pages'][:6]])
            self.assertEqual(len(pub.ledger['requests']), 7)
            self.assertEqual(helper.preserved(pub), preserved)
            # Eight primary slots remain; a due source check still proceeds.
            due = metadata.NOW + pub.policy.normal_interval_seconds
            token = pub.reserve(pub.head, due)
            self.assertIsNotNone(token)
            self.assertEqual(pub.ledger['requests'][-1]['purpose'], 'schedule')
            pub.finish_request(pub.head, token, due, 'success')
            self.assertEqual(helper.preserved(pub), preserved)

    def test_vc07_s02_negative(self):
        helper = metadata.CurrentMetadataPublisherTest()
        with helper.authority() as (pub, store, code):
            transport = metadata.CapturedTransport(pub, self.metadata, self.metadata_files)
            # Due schedule preempts catalog before any provider call.
            collector = CatalogRefresh(pub, transport=transport)
            self.assertEqual(collector.run(metadata.NOW).status, 'yielded')
            self.assertEqual(transport.calls, [])
            helper.primary(pub, metadata.NOW, accept=True)
            preserved = deepcopy(helper.preserved(pub))
            token = pub.reserve_catalog(pub.head, metadata.NOW)
            self.assertIsNotNone(token)
            reserved_head = pub.head
            stale = Publisher(store, code)
            stale.reconcile(None)
            transport.pub = stale
            with self.assertRaisesRegex(StateError, 'uncertain request'):
                CatalogRefresh(stale, transport=transport).run(metadata.NOW)
            self.assertEqual(transport.calls, [])
            self.assertEqual(sum(row['outcome'] == 'reserved' for row in stale.ledger['requests']), 1)
            self.assertEqual(helper.preserved(stale), preserved)
            self.assertEqual(stale.ledger['requests'][-1]['id'], token)
            # The original reservation has no implicit source-freshness effect.
            self.assertEqual(stale.head, reserved_head)
