(:test,:typecheck(false)) function cleanupFreshnessSharedPolicy(logger){
    var cases=[
        [998000l,1000000l,999000l,3600,900,null,3,false],
        [998000l,1000000l,1000700l,3600,900,null,3,true],
        [1000000l,2000000l,1010799l,3600,900,null,3,false],
        [1000000l,2000000l,1010800l,3600,900,null,3,true],
        [1000000l,2000000l,1000600l,3600,900,300,2,true]
    ];
    for(var i=0;i<cases.size();i+=1){
        var c=cases[i];var p={"normal_interval_seconds"=>c[3],"near_interval_seconds"=>c[4],"fast_interval_seconds"=>c[5],"stale_cycles"=>c[6]};
        var scene=new SceneContext({"id"=>"launch","precision"=>"exact","target"=>c[1],"sourceValidated"=>c[0],"freshnessPolicy"=>p});scene.advanceFreshness(c[2]);
        var reminder={"base"=>"https://fixture.invalid","id"=>"launch","sequence"=>1,"target"=>c[1],"validated"=>c[0],"since"=>c[0],"stale"=>false,"policy"=>[c[3],c[4],c[5],c[6]]};
        Test.assertEqual(scene.stale,c[7]);Test.assertEqual(LaunchReminder.valid(reminder,"https://fixture.invalid",c[2]),!c[7]);
    }
    var future={"base"=>"https://fixture.invalid","id"=>"launch","sequence"=>1,"target"=>2000000l,"validated"=>1000001l,"since"=>999000l,"stale"=>false,"policy"=>[3600,900,null,3]};
    Test.assert(!LaunchReminder.valid(future,"https://fixture.invalid",1000000l));
    var sticky=new SceneContext({"id"=>"launch","precision"=>"exact","target"=>2000000l,"sourceValidated"=>1000000l,"freshnessPolicy"=>{"normal_interval_seconds"=>3600,"near_interval_seconds"=>900,"fast_interval_seconds"=>null,"stale_cycles"=>3},"stale"=>true});sticky.advanceFreshness(1000001l);Test.assert(sticky.stale);
    logger.debug("CLEANUP_FRESHNESS normal_near_fast future_refused sticky_retained");return true;
}
