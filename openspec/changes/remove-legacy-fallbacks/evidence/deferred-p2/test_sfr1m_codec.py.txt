"""Finite independent checks of the current publisher's SFR1 codec."""
import gzip
import itertools
import unittest
from pathlib import Path

from tests.test_sfr1m_offline_current import literal_decode
from tools.spacefox import native_profile as native

ROOT = Path(__file__).resolve().parents[1]


def reference_row(row, previous, channels):
    """Literal pre-optimization token rule, independent of tuple grouping."""
    width = len(row) // channels
    x = 0
    result = bytearray()
    while x < width:
        copy = 0
        while copy < 128 and x + copy < width and row[(x+copy)*channels:(x+copy+1)*channels] == previous[(x+copy)*channels:(x+copy+1)*channels]:
            copy += 1
        color = row[x*channels:(x+1)*channels]
        run = 1
        while run < 128 and x+run < width and row[(x+run)*channels:(x+run+1)*channels] == color:
            run += 1
        if copy and copy * 4 >= run:
            result.append(127 + copy)
            x += copy
        else:
            result.append(run - 1)
            result.extend(color)
            x += run
    return bytes(result)


class CurrentSfrCodecTest(unittest.TestCase):
    def test_row_encoding_matches_independent_rule(self):
        for size in range(1, 6):
            for left in itertools.product(range(2), repeat=size):
                for above in itertools.product(range(2), repeat=size):
                    for channels in (3, 4):
                        row = b''.join(bytes([v])*channels for v in left)
                        previous = b''.join(bytes([v])*channels for v in above)
                        self.assertEqual(native._row_tokens(row, previous, channels), reference_row(row, previous, channels))
        for size in (127, 128, 129, 256, 257, 390):
            row = b''.join(bytes([int(i >= size//2)])*3 for i in range(size))
            previous = b''.join(bytes([int(i % 3 != 0)])*3 for i in range(size))
            self.assertEqual(native._row_tokens(row, previous, 3), reference_row(row, previous, 3))

    def test_captured_current_roles_roundtrip_and_bounds(self):
        for role, width, height, channels in (('map',390,390,3),('rocket',54,124,4),('siteCountry',20,12,4)):
            with self.subTest(role=role):
                pixels = gzip.decompress((ROOT/'tests/fixtures/sfr1m-current'/f'{role}.rgba.gz').read_bytes())
                chunks = native.encode_raster(pixels,width,height,channels)
                self.assertEqual(literal_decode(chunks,width,height,channels),pixels)
                with self.assertRaises(ValueError):
                    literal_decode(chunks[:-1],width,height,channels)
        with self.assertRaises(ValueError):
            native.encode_raster(b'abc',2,2,3)

    def test_current_png_owner_rejects_invalid_bytes(self):
        with self.assertRaises(ValueError):
            native.png_pixels(b'bad')


if __name__ == '__main__':
    unittest.main()
