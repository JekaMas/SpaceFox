import Toybox.Lang;
import Toybox.System;
import Toybox.Time;

// UTC arithmetic and local watch presentation have separate sources. Do not apply
// a time-zone offset to Time.now(), which is already an absolute UTC moment.
class WatchClock {
    function initialize() {}

    function utcSeconds() as Number {
        return Time.now().value();
    }

    function localTimeText() as String {
        var clock = System.getClockTime();
        return formatLocal(clock.hour, clock.min, System.getDeviceSettings().is24Hour);
    }

    function formatLocal(hour as Number, minute as Number, is24Hour as Boolean) as String {
        if (!is24Hour) {
            hour = hour % 12;
            if (hour == 0) { hour = 12; }
        }
        return hour.format("%02d") + ":" + minute.format("%02d");
    }
}
