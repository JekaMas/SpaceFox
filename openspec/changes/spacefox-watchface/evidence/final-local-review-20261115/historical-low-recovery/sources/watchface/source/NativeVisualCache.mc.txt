    }
    function maintain(){
        try{var index=store.read("sf.objects");if(!(index instanceof Array)||index.size()<92){return;}var keep=protectedObjects();var next=[];for(var i=0;i<index.size();i+=1){var r=index[i];if(r instanceof Array&&r.size()==2&&r[0] instanceof String){if(next.size()<88&&!keep.hasKey(r[0])){store.remove(objectKey(r[0]));}else{next.add(r);}}}store.write("sf.objects",next);}catch(e){}

            for(var c=0;c<v[4].size();c+=1){if(palette.indexOf(v[4][c])<0){palette.add(v[4][c]);}}for(var p=0;p<v[5].size();p+=1){if(pool.indexOf(v[5][p])<0){pool.add(v[5][p]);}}NativeMetrics.sample("logical-shard");operations+=shard.operations;refs+=shard.references;if(palette.size()>32||pool.size()>32||operations>16||refs>64){throw new Lang.InvalidValueException("Logical visual too large");}

    function activate(packet){
        if(staging==null){return false;}try{var check=new NativeSfv("[\"SFV\",1,32,[10,10],[1122867],[0,320,5243200,5242880],[[0,0,[0,1,2,3]]],[0,[]]]",null);check.animation(staging["animation"]);if(!staging["catalogDigest"].equals(packet.document["catalogDigest"])||!staging["launchId"].equals(packet.launchId())||!staging["modelId"].equals(packet.document["launch"]["configuration_id"])||!logicalDigest(staging).equals(staging["sha256"])){return false;}
            var old=store.read("sf.visual");if(old instanceof Dictionary){store.write("sf.visual.recovery",old);}store.write("sf.visual",{"v"=>1,"base"=>base,"selection"=>staging});manifest=staging;staging=null;return true;
        }catch(e){return false;}
    }
    function activateReady(packet){
        if(packet==null||packet.launchId()==null){return false;}try{var ready=store.read("sf.visual.ready");if(!(ready instanceof Dictionary)||ready["v"]!=1||!(ready["base"] instanceof String)||!ready["base"].equals(base)){return false;}staging=manifestData(ready["selection"],packet);if(activate(packet)){store.remove("sf.visual.ready");return true;}}catch(e){}return false;
    }
    function load(packet){
        manifest=null;if(packet==null||packet.launchId()==null){return null;}var keys=["sf.visual","sf.visual.recovery"];for(var i=0;i<keys.size();i+=1){try{var m=store.read(keys[i]);if(m instanceof Dictionary&&m["v"]==1&&m["base"] instanceof String&&m["base"].equals(base)){var d=manifestData(m["selection"],packet);if(logicalDigest(d).equals(d["sha256"])){manifest=d;return d;}}}catch(e){}}return null;
    }
}
