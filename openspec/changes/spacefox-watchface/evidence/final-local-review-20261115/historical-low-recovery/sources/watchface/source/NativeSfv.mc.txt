import Toybox.Lang;

(:extendedCode, :typecheck(false))
class NativeSfv {
    var value;var digest;var byteCount;var vertices=0;var operations=0;var references=0;
    function initialize(raw,expected){
        if(!(raw instanceof String)){fail();}
        var parser=new NativeJson();byteCount=parser.byteSize(raw);value=parser.parse(raw,512);validate();if(expected!=null){var h=new NativeCanonical();h.emit(value);digest=h.digest();if(!digest.equals(expected)){fail();}}
    }
    function fail(){throw new Lang.InvalidValueException("Invalid SFV1");}
    function integer(v){return v instanceof Number||v instanceof Long;}
    function bounded(v,lo,hi){return integer(v)&&v>=lo&&v<=hi;}
    function array(v,size){return v instanceof Array&&v.size()==size;}
    function validate(){
        var v=value;if(!array(v,8)||!(v[0] instanceof String)||!v[0].equals("SFV")||!bounded(v[1],1,1)||!bounded(v[2],32,32)||!array(v[3],2)||!bounded(v[3][0],1,256)||!bounded(v[3][1],1,256)){fail();}
        var palette=v[4];var pool=v[5];var commands=v[6];if(!(palette instanceof Array)||palette.size()<1||palette.size()>32||!(pool instanceof Array)||pool.size()<3||pool.size()>32||!(commands instanceof Array)||commands.size()<1||commands.size()>16){fail();}
        for(var i=0;i<palette.size();i+=1){if(!bounded(palette[i],0,16777215)||i>0&&palette[i]<=palette[i-1]){fail();}}
        var used={};for(var p=0;p<pool.size();p+=1){var point=pool[p];if(!bounded(point,0,268435455)||point%16384>v[3][0]*32||point/16384>v[3][1]*32||used.hasKey(point)){fail();}used[point]=true;}
        used={};var usedColors={};var next=0;for(var c=0;c<commands.size();c+=1){var op=commands[c];if(!array(op,3)||!bounded(op[0],0,0)||!bounded(op[1],0,palette.size()-1)||!(op[2] instanceof Array)||op[2].size()<3||op[2].size()>16){fail();}usedColors[op[1]]=true;references+=op[2].size();if(references>64){fail();}var polygon=[];
            for(var j=0;j<op[2].size();j+=1){var vi=op[2][j];if(!bounded(vi,0,pool.size()-1)){fail();}vi=vi.toNumber();if(!used.hasKey(vi)){if(vi!=next){fail();}used[vi]=true;next+=1;}polygon.add([pool[vi]%16384,pool[vi]/16384]);}
            simple(polygon);
        }
        if(next!=pool.size()||usedColors.size()!=palette.size()){fail();}vertices=pool.size();operations=commands.size();animation(v[7]);
    }
    function animation(a){
        if(!array(a,2)||!integer(a[0])||!(a[1] instanceof Array)){fail();}if(a[0]==0){if(a[1].size()!=0){fail();}return;}
        if(!bounded(a[0],2000,3000)||a[1].size()<2||a[1].size()>16){fail();}var last=-1;
        for(var i=0;i<a[1].size();i+=1){var f=a[1][i];if(!array(f,4)||!bounded(f[0],0,a[0])||f[0]<=last||!bounded(f[1],-2048,2048)||!bounded(f[2],-2048,2048)||!bounded(f[3],128,512)){fail();}if(i==0&&(f[0]!=0||f[1]!=0||f[2]!=0||f[3]!=256)){fail();}last=f[0];}if(last!=a[0]){fail();}
    }
    function cross(a,b,c){return (b[0].toLong()-a[0])*(c[1].toLong()-a[1])-(b[1].toLong()-a[1])*(c[0].toLong()-a[0]);}
    function between(a,b,c){return c[0]>=(a[0]<b[0]?a[0]:b[0])&&c[0]<=(a[0]>b[0]?a[0]:b[0])&&c[1]>=(a[1]<b[1]?a[1]:b[1])&&c[1]<=(a[1]>b[1]?a[1]:b[1]);}
    function intersects(a,b,c,d){var ac=cross(a,b,c);var ad=cross(a,b,d);var ca=cross(c,d,a);var cb=cross(c,d,b);return ac==0&&between(a,b,c)||ad==0&&between(a,b,d)||ca==0&&between(c,d,a)||cb==0&&between(c,d,b)||(ac>0)!=(ad>0)&&(ca>0)!=(cb>0);}
    function simple(points){var area=0l;var size=points.size();for(var i=0;i<size;i+=1){var a=points[i];var b=points[(i+1)%size];if(a[0]==b[0]&&a[1]==b[1]){fail();}area+=a[0].toLong()*b[1]-b[0].toLong()*a[1];for(var j=i+1;j<size;j+=1){if(j==i+1||i==0&&j==size-1){continue;}if(intersects(a,b,points[j],points[(j+1)%size])){fail();}}}if(area==0){fail();}}
}
