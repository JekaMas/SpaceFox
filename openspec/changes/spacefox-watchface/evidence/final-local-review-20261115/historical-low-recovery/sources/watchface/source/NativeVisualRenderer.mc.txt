import Toybox.Graphics;
import Toybox.Lang;

(:typecheck(false))
class NativeVisualRenderer {
    var bitmap=null;var manifest=null;var animations=null;var prepared=0;var transform=null;var drawOptions=null;
    function initialize(){transform=new Graphics.AffineTransform();drawOptions={:transform=>transform};}
    function prepare(cache){
        bitmap=null;manifest=cache.manifest;if(manifest==null){return false;}
        var ref=Graphics.createBufferedBitmap({:width=>54,:height=>124,:alphaBlending=>Graphics.ALPHA_BLENDING_FULL});var buffer=ref.get();var dc=buffer.getDc();dc.setColor(Graphics.COLOR_TRANSPARENT,Graphics.COLOR_TRANSPARENT);dc.clear();
        for(var i=0;i<manifest["shards"].size();i+=1){var v=cache.decoded[i];var scale=54.0/v[3][0];if(scale>124.0/v[3][1]){scale=124.0/v[3][1];}var ox=(54-v[3][0]*scale)/2;var oy=(124-v[3][1]*scale)/2;
            for(var j=0;j<v[6].size();j+=1){var op=v[6][j];var points=[];for(var k=0;k<op[2].size();k+=1){var packed=v[5][op[2][k].toNumber()];points.add([ox+(packed%16384).toFloat()*scale/32,oy+(packed/16384).toFloat()*scale/32]);}dc.setColor(v[4][op[1].toNumber()].toNumber(),Graphics.COLOR_TRANSPARENT);dc.fillPolygon(points);}
        }
        bitmap=buffer;animations=manifest["animation"];prepared+=1;return true;
    }
    function duration(){return animations!=null&&animations[0]>0?animations[0].toFloat()/1000.0:2.5;}
    function draw(dc,progress,ascent){
        if(bitmap==null){return;}var x=22.0;var y=118.0-ascent;var scale=1.0;
        if(animations!=null && animations[0]>0 && progress>0 && progress<1){var elapsed=progress*animations[0];var frames=animations[1];for(var i=1;i<frames.size();i+=1){if(elapsed<=frames[i][0]){var a=frames[i-1];var b=frames[i];var f=(elapsed-a[0])/(b[0]-a[0]);x+=(a[1]+(b[1]-a[1])*f)*124/(manifest["canvas"][0]*32);y+=(a[2]+(b[2]-a[2])*f)*124/(manifest["canvas"][1]*32);scale=(a[3]+(b[3]-a[3])*f)/256.0;break;}}}
        transform.setToTranslation((27-27*scale).toFloat(),(62-62*scale).toFloat());transform.scale(scale.toFloat(),scale.toFloat());
        dc.setClip(22,118-ascent,54,124);dc.drawBitmap2(x,y,bitmap,drawOptions);dc.clearClip();
    }
}
